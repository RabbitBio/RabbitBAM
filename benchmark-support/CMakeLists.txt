add_library(swbam_benchmark_support STATIC ram_file.cpp storage.cpp)
target_compile_definitions(swbam_benchmark_support PUBLIC SWBAM_ENABLE_RAM_STORAGE=1)
target_include_directories(swbam_benchmark_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(swbam_benchmark_support PUBLIC swbam::swbam)
option(SWBAM_RAM_IO_AUDIT "Audit linked user-data POSIX calls during RAM MPI interval" OFF)
if(SWBAM_RAM_IO_AUDIT)
    target_sources(swbam_benchmark_support PRIVATE io_audit.cpp)
    target_compile_definitions(swbam_benchmark_support PUBLIC SWBAM_RAM_IO_AUDIT=1)
    foreach(symbol open open64 close read write pread pwrite fopen64 fread fwrite)
        target_link_libraries(swbam_benchmark_support PUBLIC "-Wl,--wrap=${symbol}")
    endforeach()
endif()
